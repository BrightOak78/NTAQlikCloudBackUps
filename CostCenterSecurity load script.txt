///$tab משתמשים פנימים
InternalUserOrgStep1:
LOAD distinct
     NTNAME as USERNAME_OS
FROM [$(vQvdSecurityDW)Security_Budget.QVD](qvd);
//left join
inner join 
LOAD  
     [קוד משתמש] as USERNAME_OS, 
     [קוד משתמש], 
     [שם משתמש], 
     [שם משתמש פרטי], 
     [שם משתמש משפחה], 
     [שם משתמש אימייל], 
     [קוד אגף], 
     [שם אגף], 
     [קוד חטיבה], 
     [שם חטיבה], 
     [קוד קבוצת רכש], 
     [שם קבוצת רכש]
FROM [$(vQvdSecurityDW)Users_Data.QVD] (qvd)
Where [קוד חטיבה]<>'EXT';// left( [קוד חטיבה],1) <> 9;
//EXIT SCRIPT
left join
LOAD Distinct
        KOSTL as [מרכז עלות], 
        num#(VERAK) as [קוד חטיבה]
FROM [$(vLibDataFiles_RAW)/CSKS.qvd](qvd)
Where len(KOSTL) > 1;
left join
LOAD   
	 FKSTL as [מרכז עלות], 
 	 OBJNR as [מספר אובייקט_OBJNR], 
     PSPNR as NodeID
FROM [$(vLibDataFiles_RAW)/PRPS.qvd](qvd)
Where len(FKSTL) > 1;
left Join
///      Wbs אירגוני
LOAD 
     "WBS מס." as [WBS 'מס], 
     NodeID 
FROM [$(vLibDataFiles_DW)/WBS.qvd](qvd);

InternalUserOrg:
Load distinct USERNAME_OS,
              'USER' as ACCESS,
              [WBS 'מס] 
resident InternalUserOrgStep1;
drop table InternalUserOrgStep1;

store * from InternalUserOrg into [$(vLibDataFiles_DW)/InternalUserOrg.qvd];
drop table InternalUserOrg; 


InternalUserProj:
LOAD distinct
     NTNAME as USERNAME_OS
FROM [$(vQvdSecurityDW)Security_Budget.QVD](qvd);
//where Budget = 1;
inner join 
LOAD  'USER' as ACCESS,
     [קוד משתמש] as USERNAME_OS
FROM [$(vQvdSecurityDW)Users_Data.QVD] (qvd)
Where [קוד חטיבה]<>'EXT'; //left( [קוד חטיבה],1) <> 9;
join
LOAD Distinct
     "WBS מס." as [WBS 'מס]
FROM [$(vLibDataFiles_DW)/WBS.qvd](qvd)
where left([WBS מס.] ,1) = 'P';
store * from InternalUserProj into [$(vLibDataFiles_DW)/InternalUserProj.qvd];
drop table InternalUserProj;

OBJNRs:
LOAD 
	 PDOBJNR as OBJNR,
	 WBSELEMENT as [WBS 'מס]
FROM [$(vLibDataFiles_DW)/WBS.qvd](qvd);
Concatenate 
LOAD   
    KeyObjnr as OBJNR, 
    WBSELEMENT as [WBS 'מס]
FROM [$(vLibDataFiles_DW)/WBS.qvd](qvd);

ExternalUserProj:
LOAD Distinct //*,
	 'USERSEX' as QVGROUP,
 	 'USER' as ACCESS,
// 	 שותף_PARNR as PM,
     num#(PARNR) as PM,
     OBJNR
FROM [$(vLibDataFiles_RAW)/IHPA.qvd](qvd)
where  PARVW = 'D4' and KZLOESCH<>'X';
left join
LOAD *
Resident OBJNRs;
DROP Table OBJNRs;
left join
LOAD 
	 BNAME as NTNAME, 
      num#(PARVA) as PM
FROM [$(vLibDataFiles_RAW)/USR05.qvd](qvd)
Where [PARID]='LIF'
;
store * from ExternalUserProj into [$(vLibDataFiles_DW)/ExternalUserProj.qvd];
drop table ExternalUserProj;

Users:

    USERNAME_DOCGROUP as QVGROUP

LOAD Distinct
     "WBS מס." as [WBS 'מס]
FROM [$(vLibDataFiles_DW)/WBS.qvd](qvd);

LOAD  
     NTNAME, 
     ACCESS, 
     [WBS 'מס],
     QVGROUP

	NTNAME, 
    ACCESS ,
    USERNAME_DOCGROUP as ORIGINAL_DOCGROUP
