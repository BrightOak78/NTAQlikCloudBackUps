COFP:
NoConcatenate
// LOAD  *
// Resident KeyObjnr_%PSPNR;
// DROP Table KeyObjnr_%PSPNR;
// Left Join (COFP)
//	 54 – חשבונית נקלטה, עלות בפועל.
//	 57 = תשלומים בפועל (Actual Payments).
//	 60 – חשבונית בהמתנה/בטיוטה, עדיין לא עלות בפועל.	 
//	 61 = מקדמות לספקים (Down Payments).
//	 63 = התאמות/קיזוז מקדמות (Down Payment Clearing).
//	 64 = העברות/שינויים בקרנות (Fund Transfers).
LOAD 
	 Date(ZHLDT) as COFP_Date,
	 GJAHR as [שנת כספים], 
     BELNR,  //PS cash Document
     MWSKZ as [קוד מעמ], 
     LIFNR, //Account Number of Vendor or Creditor
//     [ח-ן G/L_HKONT], //General Ledger Account
     BELNR2, //Accounting Document Number - BELNR2 – מספר מסמך FI משלים/קשור (לרוב מסמך תשלום/קיזוז/התאמה בפועל
     GJAHR2, 
	 num#(EBELN)& num#(EBELP) as %EBELN_EBELP,
     num#(EBELN) as EBELN, 
     num#(EBELP) as EBELP,  //Item Number of Purchasing Document
     KOKRS, //Controlling Area
     SGTXT, //Item Text
     [KBLPOS] as KBLPOS, //Earmarked Funds: Document Item
     text(OBJNR) as KeyObjnr,
     KBLNR as [מספר מסמך_KBLNR], //Document Number for Earmarked Funds
     PERIO, 
     VRGNG, //CO Business Transaction
     BELTP, //Debit type
     WRTTP as [סוג ערך], //Value Type
	if(match(WRTTP,54,60),WKGBTR*-1,0) as [פוטנציאל תשלומים],//*-1
	 if(match(WRTTP,57,64),WKGBTR*-1,if(match(WRTTP,'61','63') and Left(BELNR2,3)='200' and match([HI_FIVOR],'90','60'),WKGBTR*-1,0)) as [תשלומים],//*-1
     POSIT, //Internal commitment item (8 chars)
     ZHLDT, //Funds Management - Update Date
     1 as SourceCOFP_flag,
     If(Match(WRTTP,57,61,63,64), 'תשלומים',If(Match(WRTTP,54,60), 'פוטנציאל תשלומים')) as SourceCOFP,
     if(WRTTP='57','תשלומים',if(WRTTP='60','מסמך בהתאמה',if(WRTTP='61','מקדמות',if(WRTTP='63','התאמות מקדמות',if(WRTTP='64','העברת קרנות'))))) as [תאור סוג ערך], 

 	 if(WRTTP='64' and (IsNull(BELNR2) or Trim(BELNR2)=''), KBLNR,BELNR2) as [מספר מסמך],     
 	 num#(if(WRTTP='64' and (IsNull(BELNR2) or Trim(BELNR2)=''), KBLNR,BELNR2)) as [מספר מסמךNUM],  
 	 if(match(WRTTP,'61','63') and Left(BELNR2,3)='200' and match([HI_FIVOR],'90','60'), 0,1) as [דגל קיזוזי מקדמות ותשלום מקדמות],
 	 if([HI_FIVOR]='60','תשלום מקדמה',if([HI_FIVOR]='90','קיזוז מקדמה')) as [תנועה פיננסית],
// 	  if(WRTTP='61' and Left([מספר מסמך_BELNR1],3)='200', 0,1) as [דגל קיזוזי מקדמות],
     GJAHR1, 
     BELNR1,  //Accounting Document Number - BELNR1 – מספר מסKBLNRמך FI מקורי שממנו נבעה התנועה (למשל חשבונית ספק).
     BUZEI2 as BUZEI1, //Number of Line Item Within Accounting Document
     num#([LO_ACCOUNT]) as KSTAR,//General Ledger Account
     if(match(WRTTP,50),WKGBTR*-1,0) as [דרישות רכש],
     if(match(WRTTP,51),WKGBTR*-1,0) as [הזמנות רכש]    
FROM [$(vLibDataFiles_RAW)/COFP.qvd](qvd)
Where Match(WRTTP,57,61,63,64,54,60)
	and WKGBTR<>0
	and OBJNR<>'PR00001933' //PR00001933 - סגירת תקופה הוצאות
	and OBJNR<>'PR00001935' //PR00001935 - סגירת תקופה הכנסות
	and OBJNR<>'OR000000600000' 
 ;

ZZN_Hierarchy_Base:
LOAD
    text(CODE_NUMBER) as ZZN_CODE_NUMBER,
    DESCRIPTION as ZZN_CODE_NUMBER_DESCRIPTION,
//    len(replace(CODE_NUMBER, '.', '')) as LenCheck2,
    // רמת ההיררכיה (מספר הנקודות + 1)
    len(CODE_NUMBER) - len(replace(CODE_NUMBER, '.', '')) + 1 as HierarchyLevel,
//    // ParentID
	text(if(index(CODE_NUMBER, '.') > 0,
        left(CODE_NUMBER, len(CODE_NUMBER) - len(subfield(CODE_NUMBER, '.', -1)) - 1),
        null()
    )) as ParentID
FROM [$(vLibDataFiles_RAW)/ZNPS_METROBI_COT.qvd] (qvd)    
WHERE SPRAS = 'B';

ZZN_Hierarchy_t:
Hierarchy(ZZN_CODE_NUMBER, ParentID, ZZN_CODE_NUMBER_DESCRIPTION, NodePath, '/', Depth)
//Hierarchy(New_CODE_NUMBER, ParentID, ZZN_CODE_NUMBER_DESCRIPTION, ParentName, NodeName, PathName, '\', Depth)
LOAD 
	ZZN_CODE_NUMBER, 
	ParentID,
    ZZN_CODE_NUMBER_DESCRIPTION
RESIDENT ZZN_Hierarchy_Base;
Left Join
LOAD *
RESIDENT ZZN_Hierarchy_Base; 
DROP TABLE ZZN_Hierarchy_Base;

/********************************************************************
  2) RawTable: קודם מייצרים שדות "מובטחים טקסט" + KeyBase
     (בהנחה שכל הערכים מלאים, אין טיפול ב-Null)
*********************************************************************/
RawTableTemp:
NoConcatenate
LOAD
    Key,
    LineCode,
    WorkTypeCode,
    WorkPackageCode,
    NBS1,
    NBS2,
    Networking,
    Networking & '|'   & LineCode & '|' & WorkTypeCode & '|' & WorkPackageCode & '|' & NBS1 & '|' & NBS2 as KeyBase
;
LOAD
   // RecNo() as RowID,
    Key,
    Trim(Networking & '')            as Networking,
    Trim([Line Code] & '')            as LineCode,
    Trim([Work Type Code] & '')       as WorkTypeCode,
    Trim([Work package Code] & '')    as WorkPackageCode,
    Trim([NBS LEVEL 1 Code] & '')     as NBS1,
    Trim([NBS LEVEL 2 Code] & '')     as NBS2
Resident Src;

DROP Table Src;


//**


left join 

LOAD DISTINCT
    Key,

    // שדה הבחירה היחיד: ערכי KeyBase "מרופדים" לפי רמה
    If(IterNo()=1,
        SubField(KeyBase,'|',1) & Repeat('|',5),

    If(IterNo()=2,
        SubField(KeyBase,'|',1) & '|' & SubField(KeyBase,'|',2) & Repeat('|',4),

    If(IterNo()=3,
        SubField(KeyBase,'|',1) & '|' & SubField(KeyBase,'|',2) & '|' & SubField(KeyBase,'|',3) & Repeat('|',3),

    If(IterNo()=4,
        SubField(KeyBase,'|',1) & '|' & SubField(KeyBase,'|',2) & '|' & SubField(KeyBase,'|',3) & '|' & SubField(KeyBase,'|',4) & Repeat('|',2),

    If(IterNo()=5,
        SubField(KeyBase,'|',1) & '|' & SubField(KeyBase,'|',2) & '|' & SubField(KeyBase,'|',3) & '|' & SubField(KeyBase,'|',4) & '|' & SubField(KeyBase,'|',5) & Repeat('|',1),

        // IterNo()=6 => full
        KeyBase
    ))))) as KeyBase_Select,

    IterNo() as LevelOrder

Resident RawTableTemp
While Len(Trim(KeyBase))>0
  and IterNo() <= 6;


RawTable:
Load *,
  RecNo() as RowID
resident RawTableTemp;
drop table RawTableTemp; 

/********************************************************************
  3) Reverse: הופכים את KeyBase תו-תו
*********************************************************************/
Chars:
LOAD
    RowID,
    IterNo() as Idx,
    Mid(KeyBase_Select, Len(KeyBase_Select) - IterNo() + 1, 1) as Ch
Resident RawTable
While IterNo() <= Len(KeyBase_Select);

/********************************************************************
  4) Key_Reversed: מאגדים חזרה למחרוזת
*********************************************************************/
KeyRev:
NoConcatenate
LOAD
    RowID,
    Concat(Ch, '', Idx) as Key_Reversed
Resident Chars
Group By RowID;

DROP Table Chars;

/********************************************************************
  5) Key_Encrypted: החלפות אחרי ההיפוך בלבד
     M = Y3PLD
     W = 4RZZ
     IN = 7UP   (כל הופעה של IN)
*********************************************************************/
Enc:
NoConcatenate
LOAD
    RowID,
    Key_Reversed,
    Replace(
        Replace(
            Replace(Key_Reversed, 'IN', '7UP'),
            'M', 'Y3PLD'
        ),
        'W', '4RZZ'
    ) as %Key_Encrypted
Resident KeyRev;

DROP Table KeyRev;

/********************************************************************
  6) Join + Final
*********************************************************************/
Left Join (RawTable)
LOAD
    RowID,
    Key_Reversed,
    %Key_Encrypted
Resident Enc;

DROP Table Enc;

COFP:
NoConcatenate
LOAD  *
Resident KeyObjnr_%PSPNR;

Left Join (COFP)
//	 54 – חשבונית נקלטה, עלות בפועל.
//	 57 = תשלומים בפועל (Actual Payments).
//	 60 – חשבונית בהמתנה/בטיוטה, עדיין לא עלות בפועל.	 
//	 61 = מקדמות לספקים (Down Payments).
//	 63 = התאמות/קיזוז מקדמות (Down Payment Clearing).
//	 64 = העברות/שינויים בקרנות (Fund Transfers).
LOAD 
	 Date(ZHLDT) as COFP_Date,
	 GJAHR as [שנת כספים], 
     BELNR,  //PS cash Document
     MWSKZ as [קוד מעמ], 
     LIFNR, //Account Number of Vendor or Creditor
//     [ח-ן G/L_HKONT], //General Ledger Account
     BELNR2, //Accounting Document Number - BELNR2 – מספר מסמך FI משלים/קשור (לרוב מסמך תשלום/קיזוז/התאמה בפועל
     GJAHR2, 
	 num#(EBELN)& num#(EBELP) as %EBELN_EBELP,
     num#(EBELN) as EBELN, 
     num#(EBELP) as EBELP,  //Item Number of Purchasing Document
     KOKRS, //Controlling Area
     SGTXT, //Item Text
     [KBLPOS] as KBLPOS, //Earmarked Funds: Document Item
     text(OBJNR) as KeyObjnr,
     KBLNR as [מספר מסמך_KBLNR], //Document Number for Earmarked Funds
     PERIO, 
     VRGNG, //CO Business Transaction
     BELTP, //Debit type
     WRTTP as [סוג ערך], //Value Type
	if(match(WRTTP,54,60),WKGBTR*-1,0) as [פוטנציאל תשלומים],//*-1
	 if(match(WRTTP,57,64),WKGBTR*-1,if(match(WRTTP,'61','63') and Left(BELNR2,3)='200' and match([HI_FIVOR],'90','60'),WKGBTR*-1,0)) as [תשלומים],//*-1
     POSIT, //Internal commitment item (8 chars)
     ZHLDT, //Funds Management - Update Date
     1 as SourceCOFP_flag,
     If(Match(WRTTP,57,61,63,64), 'תשלומים',If(Match(WRTTP,54,60), 'פוטנציאל תשלומים')) as SourceCOFP,
     if(WRTTP='57','תשלומים',if(WRTTP='60','מסמך בהתאמה',if(WRTTP='61','מקדמות',if(WRTTP='63','התאמות מקדמות',if(WRTTP='64','העברת קרנות'))))) as [תאור סוג ערך], 

 	 if(WRTTP='64' and (IsNull(BELNR2) or Trim(BELNR2)=''), KBLNR,BELNR2) as [מספר מסמך],     
 	 num#(if(WRTTP='64' and (IsNull(BELNR2) or Trim(BELNR2)=''), KBLNR,BELNR2)) as [מספר מסמךNUM],  
 	 if(match(WRTTP,'61','63') and Left(BELNR2,3)='200' and match([HI_FIVOR],'90','60'), 0,1) as [דגל קיזוזי מקדמות ותשלום מקדמות],
 	 if([HI_FIVOR]='60','תשלום מקדמה',if([HI_FIVOR]='90','קיזוז מקדמה')) as [תנועה פיננסית],
// 	  if(WRTTP='61' and Left([מספר מסמך_BELNR1],3)='200', 0,1) as [דגל קיזוזי מקדמות],
     GJAHR1,

     BUZEI2 as BUZEI1, //Number of Line Item Within Accounting Document
