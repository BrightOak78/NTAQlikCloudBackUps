Security_Line_Primavera:
//Authenticated_Project:
LOAD Distinct
	 DOCGROUP as USERNAME_DOCGROUP
FROM [lib://API & Connections:Sharepoint365/BIDB/Security/Security_AddOn.xlsx]
(ooxml, embedded labels, table is LinesGroups)
Where Privavera_Line_Code='*';
Left Join
LOAD Distinct 
	 Privavera_Line_Code as [Line Code]
FROM [lib://API & Connections:Sharepoint365/BIDB/Security/Security_AddOn.xlsx]
(ooxml, embedded labels, table is LinesGroups)
Where  Privavera_Line_Code<>'*' and not IsNull(Privavera_Line_Code) ;

Concatenate
LOAD Distinct
	 DOCGROUP as USERNAME_DOCGROUP, 
     Privavera_Line_Code as [Line Code]
FROM [lib://API & Connections:Sharepoint365/BIDB/Security/Security_AddOn.xlsx]
(ooxml, embedded labels, table is LinesGroups)
Where Privavera_Line_Code<>'*' and not IsNull(Privavera_Line_Code);


STORE [Security_Line_Primavera] INTO [$(vLibDataFiles_RAW)/Security_Line_Primavera.qvd](qvd);

drop Table Security_Line_Primavera;
//Exit Script
