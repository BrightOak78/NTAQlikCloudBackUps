United_WH_NEW:
CrossTable (Project, TotalManHours, 2)
LOAD
    Num(
        Year(Date#(Replace(תאריך,'.','/'),'D/M/YYYY')) * 100 +
        Month(Date#(Replace(תאריך,'.','/'),'D/M/YYYY'))
    ) as YYYYMM,

    Date#(Replace(תאריך,'.','/'),'D/M/YYYY') as WorkDate,
     "קו אדום",
    "קו ירוק",
    "קו סגול"
FROM [lib://API & Connections:Sharepoint365/BIDB/בטיחות SF/שעות עבודה קבלנים בכלל הקוים.xlsx]
(ooxml, embedded labels, table is Sheet1);

// ============================================================
// 3. סיכום שעות לפי חודש
// ============================================================

ProjectHoursMonthly:
NoConcatenate
LOAD
    ProjectID,
    Project,
    YYYYMM,
    Sum(Num#(TotalManHours)) as MonthlyManHours
Resident United_WH_NEW
Group By
    ProjectID,
    Project,
    YYYYMM;

Drop Table United_WH_NEW;


// ============================================================
// 4. מציאת חודש ראשון לכל פרויקט
// ============================================================

ProjectStartMonth:
LOAD
    ProjectID,
    Project,
    Min(YYYYMM) as StartYYYYMM
Resident ProjectHoursMonthly
Group By
    ProjectID,
    Project;


// ============================================================
// 5. יצירת כל החודשים מתחילת הפרויקט ועד החודש הנוכחי
// ============================================================

ProjectMonths:
NoConcatenate
LOAD
    ProjectID,
    Project,

    Num(
        Year(AddMonths(StartMonthDate, IterNo()-1)) * 100 +
        Month(AddMonths(StartMonthDate, IterNo()-1))
    ) as YYYYMM

While AddMonths(StartMonthDate, IterNo()-1) <= MonthStart(Today());

LOAD
    ProjectID,
    Project,
    MakeDate(
        Floor(StartYYYYMM / 100),
        Mod(StartYYYYMM,100),
        1
    ) as StartMonthDate
Resident ProjectStartMonth;

Drop Table ProjectStartMonth;


// ============================================================
// 6. חיבור נתוני השעות החודשיים
// ============================================================

Left Join (ProjectMonths)

LOAD
    ProjectID,
    YYYYMM,
    MonthlyManHours
Resident ProjectHoursMonthly;

Drop Table ProjectHoursMonthly;


// ============================================================
// 7. הפיכת חודשים חסרים ל-0
// ============================================================

ProjectHoursComplete:
NoConcatenate
LOAD
    ProjectID,
    Project,
    YYYYMM,

    Alt(MonthlyManHours,0) as MonthlyManHours

Resident ProjectMonths;

Drop Table ProjectMonths;


// ============================================================
// 8. חישוב מצטבר
// ============================================================

ProjectHoursAccumulated:
NoConcatenate
LOAD
    ProjectID,
    Project,
    YYYYMM,

    MonthlyManHours,

    If(
        ProjectID = Previous(ProjectID),
        RangeSum(
            Peek('AccumulatedManHours'),
            MonthlyManHours
        ),
        MonthlyManHours
    ) as AccumulatedManHours

Resident ProjectHoursComplete
Order By
    ProjectID,
    YYYYMM;

Drop Table ProjectHoursComplete;

LTITemp:
LOAD
    projectId as ProjectID,

    Num(
        Year(Date(Date,'DD/MM/YYYY')) * 100 +
        Month(Date(Date,'DD/MM/YYYY'))
    ) as LTI_YearMonth,

    If(
        [Injury Specification] = 'LTI - Lost Time Injury',
        1,
        0
    ) as Damage_LTI_Flag

FROM [$(vLibDataFiles_RAW)/IncAccRep_View.qvd]
(qvd)
where (match(projectId,'268440770','268440772') and Phase = 'Infra1') or not match(projectId,'268440770','268440772');


// ============================================================
// 2. סיכום אירועים לפי חודש
// ============================================================

LTIMonthly:
NoConcatenate
LOAD
    ProjectID,
    LTI_YearMonth,

    Sum(Damage_LTI_Flag) as LTI_InjuryLtiCountPerMonth

Resident LTITemp
Group By
    ProjectID,
    LTI_YearMonth;

Drop Table LTITemp;


// ============================================================
// 3. חודש ראשון לכל פרויקט
// ============================================================

LTIProjectStart:
LOAD
    ProjectID,
    Min(LTI_YearMonth) as StartYYYYMM
Resident LTIMonthly
Group By ProjectID;


// ============================================================
// 4. יצירת כל החודשים עד החודש הנוכחי
// ============================================================

LTIProjectMonths:
NoConcatenate
LOAD
    ProjectID,

    Year(AddMonths(StartMonthDate,IterNo()-1)) as LTI_Year,

    Num(
        Month(AddMonths(StartMonthDate,IterNo()-1)),
        '00'
    ) as LTI_Month,

    Num(
        Year(AddMonths(StartMonthDate,IterNo()-1)) * 100 +
        Month(AddMonths(StartMonthDate,IterNo()-1))
    ) as LTI_YearMonth

While AddMonths(StartMonthDate,IterNo()-1) <= MonthStart(Today());

LOAD
    ProjectID,

    MakeDate(
        Floor(StartYYYYMM/100),
        Mod(StartYYYYMM,100),
        1
    ) as StartMonthDate

Resident LTIProjectStart;

Drop Table LTIProjectStart;


// ============================================================
// 5. חיבור נתוני ה-LTI
// ============================================================

Left Join (LTIProjectMonths)

LOAD
    ProjectID,
    LTI_YearMonth,
    LTI_InjuryLtiCountPerMonth
Resident LTIMonthly;

Drop Table LTIMonthly;


// ============================================================
// 6. חודשים ללא אירוע = 0

LTIComplete:
NoConcatenate
LOAD
    ProjectID,
    LTI_Year,
    LTI_Month,
    LTI_YearMonth,

    Alt(LTI_InjuryLtiCountPerMonth,0)
        as LTI_InjuryLtiCountPerMonth

Resident LTIProjectMonths;

Drop Table LTIProjectMonths;


// ============================================================
// 7. חישוב LTI מצטבר
// ============================================================

LTIAccYYYYMM:
NoConcatenate
LOAD
    ProjectID,
    LTI_Year,
    LTI_Month,
    LTI_YearMonth,

    LTI_InjuryLtiCountPerMonth,

    If(
        ProjectID = Previous(ProjectID),
        RangeSum(
            Peek('LTI_InjuryLtiCountAccumulated'),
            LTI_InjuryLtiCountPerMonth
        ),
        LTI_InjuryLtiCountPerMonth
    ) as LTI_InjuryLtiCountAccumulated

Resident LTIComplete
Order By
    ProjectID,
    LTI_YearMonth;

Drop Table LTIComplete;
