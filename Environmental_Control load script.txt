Hierarchy_temp1:
NoConcatenate
LOAD
    Key,
    [Line Code],
    [Work package Code],
    [NBS LEVEL 1 Code],
    [NBS LEVEL 2 Code],
    [Work Type Code],

    [Work package],
    [NBS LEVEL 1],
    [NBS LEVEL 2],
    [Work Type],

    [Line Code] & '|' & [Work package Code] & '|' & [NBS LEVEL 1 Code] & '|' & [NBS LEVEL 2 Code] & '|' & [Work Type Code] as KeyBase

Resident Hierarchy_temp;
DROP Table Hierarchy_temp;

/*******************************************************
  יצירת KeyBase_Select לפי כל רמה בהיררכיה
*******************************************************/

Left Join (Hierarchy_temp1)
LOAD DISTINCT
    Key,

    If(IterNo() = 1,
        SubField(KeyBase, '|', 1) & Repeat('|', 4),

    If(IterNo() = 2,
        SubField(KeyBase, '|', 1) & '|' &
        SubField(KeyBase, '|', 2) & Repeat('|', 3),

    If(IterNo() = 3,
        SubField(KeyBase, '|', 1) & '|' &
        SubField(KeyBase, '|', 2) & '|' &
        SubField(KeyBase, '|', 3) & Repeat('|', 2),

    If(IterNo() = 4,
        SubField(KeyBase, '|', 1) & '|' &
        SubField(KeyBase, '|', 2) & '|' &
        SubField(KeyBase, '|', 3) & '|' &
        SubField(KeyBase, '|', 4) & Repeat('|', 1),

        // IterNo() = 5
        KeyBase
    )))) as KeyBase_Select,

    IterNo() as LevelOrder

Resident Hierarchy_temp1
While Len(Trim(KeyBase)) > 0
  and IterNo() <= 5;
  
Hierarchy_temp2:
NoConcatenate
LOAD
    *,
    RecNo() as RowID
Resident Hierarchy_temp1;

DROP Table Hierarchy_temp1;
