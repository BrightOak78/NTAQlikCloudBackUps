///$tab 10. Movement
// =====================================================
// NCR Monthly Backlog by Line
// =====================================================





// // 1. תנועות פתיחה
NCR_Movements:
LOAD distinct
    "Reference Number ID",
    Line,
    MonthStart([Date Opened]) as NCR_Date,
    1 as Open_Qty,
    0 as Close_Qty
FROM [$(vLibDataFiles_DW)/NcrLogic.qvd]
(qvd)
 Where Len(Trim( [Date Opened])) > 0;


// // 2. תנועות סגירה
Concatenate (NCR_Movements)
LOAD distinct
    "Reference Number ID",
    Line,
     MonthStart([Date Closed])  as NCR_Date,
   0 as Open_Qty,
    1 as Close_Qty
FROM [$(vLibDataFiles_DW)/NcrLogic.qvd]
(qvd)
Where Len(Trim([Date Closed])) > 0;



// // 3. סיכום חודשי לפי Line
NCR_Monthly_Summary:
LOAD
    Line,
    
    NCR_Date as  NCR_Date_backlog,
    Sum(Open_Qty) as NCR_Opened,
    Sum(Close_Qty) as NCR_Closed,
    Sum(Open_Qty) - Sum(Close_Qty) as NCR_Net_Change
Resident NCR_Movements
Group By
    Line,
    NCR_Date;


// // 4. מלאי מצטבר לפי Line
NCR_Monthly_Backlog:
NoConcatenate
LOAD
    Line,
   Year(NCR_Date_backlog)&num(Month(NCR_Date_backlog),'00') as YYYMM_Backlog,
    NCR_Date_backlog,
    NCR_Opened,
    NCR_Closed,
    NCR_Net_Change,

    If(
        Line = Previous(Line),
       RangeSum(Peek('NCR_Open_Backlog'), NCR_Net_Change),
        NCR_Net_Change
    ) as NCR_Open_Backlog

Resident NCR_Monthly_Summary
Order By
    Line,
    NCR_Date_backlog;


// // 5. ניקוי טבלאות ביניים
 Drop Table NCR_Movements;
Drop Table NCR_Monthly_Summary;

  and IterNo() <= 6;
  
RawTable:
NoConcatenate
LOAD
    *,
    RecNo() as RowID
Resident RawTableTemp;
DROP Table RawTableTemp;

/********************************************************************
  3) Reverse: הופכים את KeyBase תו-תו
*********************************************************************/
Chars:
LOAD
    RowID,
    IterNo() as Idx,
    Mid(KeyBase_Select, Len(KeyBase_Select) - IterNo() + 1, 1) as Ch
Resident RawTable
While IterNo() <= Len(KeyBase_Select);

/********************************************************************
  4) Key_Reversed: מאגדים חזרה למחרוזת
*********************************************************************/
KeyRev:
NoConcatenate
LOAD
    RowID,
    Concat(Ch, '', Idx) as Key_Reversed
Resident Chars
Group By RowID;

DROP Table Chars;

/********************************************************************
  5) Key_Encrypted: החלפות אחרי ההיפוך בלבד
     M = Y3PLD
     W = 4RZZ
     IN = 7UP   (כל הופעה של IN)
*********************************************************************/
Enc:
NoConcatenate
LOAD
    RowID,
    Key_Reversed,
    Replace(
        Replace(
            Replace(Key_Reversed, 'IN', '7UP'),
            'M', 'Y3PLD'
        ),
        'W', '4RZZ'
    ) as %Key_Encrypted
Resident KeyRev;

DROP Table KeyRev;

*********************************************************************/

/********************************************************/

Left Join (RawTable)
LOAD
    RowID,
    Key_Reversed,
    %Key_Encrypted
Resident Enc;
