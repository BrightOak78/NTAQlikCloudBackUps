MMIS_1:
LOAD
    "Service Request", // as [קריאת שירות],
    "Request Date",
    
Date(Floor([Request Date]), 'DD/MM/YYYY') AS Date,
    Status, //  as סטטוס
    Source, //  as [מקור הקריאה]
    If(Department='MAINT', 'TEVEL',Department) as filter_Department,
    Department, //  as קבלן
    "Service Category" as  "תת מערכת", //  as [קטגורית הקריאה]
    "Service Code", //  as [קוד קריאה]
    "Service Code Description", //  as [תיאור סוג קריאה]
    "Type", //  as [סוג הקריאה]
    Remarks, // as הערות
    Description , // as [תיאור התקלה]
    "CMDB Equipment Code" , // as [קוד הציוד התקול CMDB]
    Equipment, // as [קוד הציוד התקול MMIS]
    "Equipment Description", //  as [תיאור הציוד התקול]
    "Work Order" , // as [פקודת עבודה]
    "WO Status", //  as [סטטוס פקודת עבודה]
    "Related To KPI", //  as [מספר מדד שירות]
    "KPI Number",
    Location as location_excel,
//    "Location Description",
    Priority, //  as [חומרת הקריאה]
    "Finish Work Date" , // as [תאריך סיום טיפול בקריאה]
    Date(Floor([Finish Work Date]), 'DD/MM/YYYY') AS finish_date,
    Department&"KPI Number" as keySLA,
    IF(
    Match(Source, 'AFC', 'NMS', 'MSS', 'Publisher Server (ROST)', 'FSCADA', 'IMS', 'PSCADA'),
    'Automatic',
    'Manual'
) AS Call_Type,

// חישוב תקלה פתוחה או סגורה
   IF( Match( Upper(Trim(Status)), 'OPEN', 'ONGOING' ) AND NOT Match( Upper(Trim([WO Status])), 'COMPLETED', 'FINISH WORK') , 1,  0) as is_open_flag,
  IF( Match( Upper(Trim([WO Status])), 'COMPLETED', 'FINISH WORK'),
        'הושלמה', 
     IF(
        Match(Upper(Trim(Status)), 'OPEN', 'ONGOING'),
       'פתוחה', 
        'בוטלה')) as status_desc,

// חישוב זמן טיפול בתקלה רק לקריאות סגורות
	Round(IF(
    	Match( Upper(Trim([WO Status])), 'COMPLETED', 'FINISH WORK'),
    	if ( [Finish Work Date]< [Request Date], 0 ,( [Finish Work Date] - [Request Date] ) * 24 )),0.01) AS [משך טיפול בתקלה בעשרוני שעות],
	Round(IF(
    	Match( Upper(Trim([WO Status])), 'COMPLETED', 'FINISH WORK'),
    	if ( [Finish Work Date]< [Request Date], 0 , ( [Finish Work Date] - [Request Date] ) )),0.01) AS [משך טיפול בתקלה בעשרוני ימים],    
  // חישוב קוד למיקום
  Upper(Trim(PurgeChar(   
    IF(WildMatch(Location, '*TTR_R*'),
        'TTR_' & SubField(SubField(Location, 'TTR_R', 2), '-', 1),
    IF(WildMatch(Location, '*ST_R*'),
        'ST_R' & SubField(SubField(Location, 'ST_R', 2), '-', 1),        
    IF(WildMatch(Location, 'RL-DP-*'), SubField(Location, '-', 1) & '-' &  SubField(Location, '-', 2) & '-' & SubField(Location, '-', 3),
    IF(WildMatch(Location, 'RL-MV-V*'), 'RL-MV-'&SubField(Location, '-',  3),
    IF(WildMatch(Location, 'OF-SCC*'), 'OF-SCC',
       Location
    ))))) , Chr(160)))) AS [code location],
    
    SubField("Service Code", '-', 1) AS "Service code starts with"

FROM [$(vLibDataFiles_RAW)/MMIS_db.qvd] (qvd);

left join 
LOAD
    Department&"KPI Number" as keySLA,
    "דחיפות תקלה תבל",
    "SLA hours"
FROM [lib://API & Connections:Sharepoint365/BIDB/תחזוקה/addons/addons MMIS.xlsx]
(ooxml, embedded labels, table is [SLA תבל]);


Left Join
LOAD
	Upper(Trim(PurgeChar("code location", Chr(160)))) as "code location",
    "Location" ,
    Asset
FROM [lib://API & Connections:Sharepoint365/BIDB/תחזוקה/addons/addons MMIS.xlsx]
(ooxml, embedded labels, table is Location);

Left Join
LOAD
    "תת מערכת" ,
    מערכת
FROM [lib://API & Connections:Sharepoint365/BIDB/תחזוקה/addons/addons MMIS.xlsx]
(ooxml, embedded labels, table is Category);

Left Join
LOAD
    "Service code starts with",
    "AFC Item"
FROM [lib://API & Connections:Sharepoint365/BIDB/תחזוקה/addons/addons MMIS.xlsx]
(ooxml, embedded labels, table is AFC);

MMIS_3:
Load
	*,
    if ( not IsNull("SLA hours") ,  Round(if ( [זמן השבתה בשעות] > "SLA hours",  [זמן השבתה בשעות] - "SLA hours"),0.01), Null()) as hour_exception_duration  ,
   if ( status_desc='בוטלה', Null() , if(  not IsNull("SLA hours") , if ( [זמן השבתה בשעות] > "SLA hours", 1,0))) as flag_harig_tevel,
   if ( not IsNull("SLA hours") and status_desc='בוטלה',
   'לא רלוונטי' , if(  not IsNull("SLA hours") , if ( [זמן השבתה בשעות] > "SLA hours", 'חריג'
   ,'לא חריג'))) as desc_flag_harig_tevel
Resident MMIS_2;
Drop Table MMIS_2;

NoConcatenate
MMIS_4:
Load
	*,
    Call_Type as Call_Type_for_filter,
    status_desc as [סטטוס קריאה],
    IF( [זמן השבתה בשעות] <=  24 ,'Day',
    	IF( [זמן השבתה בשעות]  <=  24* 7, 'Week',
			IF([זמן השבתה בשעות] <= 24* 30 , 'Month',
    		if ( [זמן השבתה בשעות]> 30*24 ,	'Over 1 Month') ))) AS hashbata_group_time,
    if ( not IsNull("SLA hours") ,IF( hour_exception_duration <= 24 ,'Day',
    	IF(hour_exception_duration <= 24*7, 'Week',
			IF(hour_exception_duration <= 24*30, 'Month',
    			'Over 1 Month'  ))), Null()) AS tevel_hariga_group,
    if (Date >= Date#('01/06/2025','DD/MM/YYYY'),  If( Department = 'MAINT' and [KPI Number] = 'KPI_151' and flag_harig_tevel = 1,
        2500 + 500 * hour_exception_duration,
	If(Department = 'MAINT' and [KPI Number] = 'KPI_152' and flag_harig_tevel = 1,
        1000 + 500 * (hour_exception_duration/24),  0 ))) as tevel_fine_amount 
  	
Resident MMIS_3;
Drop Table MMIS_3;
